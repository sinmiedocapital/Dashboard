pivLow  = ta.pivotlow(low,  structLen, structLen)

var float  swingHigh    = na
var float  swingLow     = na
var int    swingHighBar = na
var int    swingLowBar  = na
var string mktStr       = "neutral"
if not na(pivHigh)
    swingHigh    := pivHigh
    swingHighBar := bar_index[structLen]
if not na(pivLow)
    swingLow    := pivLow
    swingLowBar := bar_index[structLen]

chLen   = math.max(10, structLen * 6)
upperCh = ta.linreg(high, chLen, 0) + atrVal * 0.3
lowerCh = ta.linreg(low,  chLen, 0) - atrVal * 0.3
plot(showChannel ? upperCh : na, "Upper Channel",
     color=color.new(#4fc3f7, 10), linewidth=2)
plot(showChannel ? lowerCh : na, "Lower Channel",
     color=color.new(#ff6b6b, 10), linewidth=2)

chBreakBull  = showChannel and ta.crossover(close,  upperCh)
chBreakBear  = showChannel and ta.crossunder(close, lowerCh)
chAboveUpper = showChannel and close > upperCh
chBelowLower = showChannel and close < lowerCh

bosBull   = not na(swingHigh) and close > swingHigh and close[1] <= swingHigh and mktStr == "bullish"
chochBull = not na(swingHigh) and close > swingHigh and close[1] <= swingHigh and (mktStr == "bearish" or mktStr == "neutral")
bosBear   = not na(swingLow)  and close < swingLow  and close[1] >= swingLow  and mktStr == "bearish"
chochBear = not na(swingLow)  and close < swingLow  and close[1] >= swingLow  and (mktStr == "bullish" or mktStr == "neutral")

snapSwingHigh    = swingHigh
snapSwingHighBar = swingHighBar
snapSwingLow     = swingLow
snapSwingLowBar  = swingLowBar

// ── Order Blocks ─────────────────────────────────────────────────────────────
var float[] obTop     = array.new_float(0)
var float[] obBot     = array.new_float(0)
var int[]   obBar     = array.new_int(0)
var bool[]  obBull    = array.new_bool(0)
var bool[]  obBreaker = array.new_bool(0)
var box[]   obBoxRefs = array.new<box>(0)

obScanLen = 30

if chochBull or bosBull
    sz = array.size(obTop)
    if sz > 0
        for i = sz - 1 to 0
            _bull = array.get(obBull,    i)
            _brk  = array.get(obBreaker, i)
            if _bull == _brk
                box.delete(array.get(obBoxRefs, i))
                array.remove(obBoxRefs, i)
                array.remove(obTop,     i)
                array.remove(obBot,     i)
                array.remove(obBar,     i)
                array.remove(obBull,    i)
                array.remove(obBreaker, i)
    mktStr    := "bullish"
    swingHigh := na
    for k = 1 to obScanLen
        if close[k] < open[k]
            obH = obUseWicks ? high[k]  : math.max(open[k], close[k])
            obL = obUseWicks ? low[k]   : math.min(open[k], close[k])
            array.unshift(obTop,     obH)
            array.unshift(obBot,     obL)
            array.unshift(obBar,     bar_index - k)
            array.unshift(obBull,    true)
            array.unshift(obBreaker, false)
            newBox = showOBs ? box.new(
                 bar_index - k, obH, bar_index, obL,
                 bgcolor=colOBBull, border_color=colOBBrdBull,
                 border_width=1, border_style=line.style_solid) : na
            array.unshift(obBoxRefs, newBox)
            if array.size(obTop) > maxOBs
                array.pop(obTop)
                array.pop(obBot)
                array.pop(obBar)
                array.pop(obBull)
                array.pop(obBreaker)
                box.delete(array.pop(obBoxRefs))
            break

if chochBear or bosBear
    sz = array.size(obTop)
    if sz > 0
        for i = sz - 1 to 0
            _bull = array.get(obBull,    i)
            _brk  = array.get(obBreaker, i)
            if _bull != _brk
                box.delete(array.get(obBoxRefs, i))
                array.remove(obBoxRefs, i)
                array.remove(obTop,     i)
                array.remove(obBot,     i)
                array.remove(obBar,     i)
                array.remove(obBull,    i)
                array.remove(obBreaker, i)
    mktStr   := "bearish"
    swingLow := na
    for k = 1 to obScanLen
        if close[k] > open[k]
            obH = obUseWicks ? high[k]  : math.max(open[k], close[k])
            obL = obUseWicks ? low[k]   : math.min(open[k], close[k])
            array.unshift(obTop,     obH)
            array.unshift(obBot,     obL)
            array.unshift(obBar,     bar_index - k)
            array.unshift(obBull,    false)
            array.unshift(obBreaker, false)
            newBox = showOBs ? box.new(
                 bar_index - k, obH, bar_index, obL,
                 bgcolor=colOBBear, border_color=colOBBrdBear,
                 border_width=1, border_style=line.style_solid) : na
            array.unshift(obBoxRefs, newBox)
            if array.size(obTop) > maxOBs
                array.pop(obTop)
                array.pop(obBot)
                array.pop(obBar)
                array.pop(obBull)
                array.pop(obBreaker)
                box.delete(array.pop(obBoxRefs))
            break

// ── Liquidity Sweep ───────────────────────────────────────────────────────────

    for j = 1 to liqLookback
        if low[j] < swingLow and close[j] > swingLow
            bullLiqSweep := true
            break
if useLiq and not na(swingHigh)
    for j = 1 to liqLookback
        if high[j] > swingHigh and close[j] < swingHigh
            bearLiqSweep := true
            break

// ── OB Zone Check ─────────────────────────────────────────────────────────────
inBullOB    = false
inBearOB    = false
bullOBSlRef = float(na)
bearOBSlRef = float(na)

nOB = array.size(obTop)
if nOB > 0
    for i = nOB - 1 to 0
        topV      = array.get(obTop,     i)
        botV      = array.get(obBot,     i)
        isBull    = array.get(obBull,    i)
        isBreaker = array.get(obBreaker, i)

        if not isBreaker
            if isBull and close < botV
                array.set(obBreaker, i, true)
                b = array.get(obBoxRefs, i)
                if not na(b)
                    box.set_bgcolor(b,      colBreaker)
                    box.set_border_color(b, colBreakerBrd)
            else if not isBull and close > topV
                array.set(obBreaker, i, true)
                b = array.get(obBoxRefs, i)
                if not na(b)
                    box.set_bgcolor(b,      colBreaker)
                    box.set_border_color(b, colBreakerBrd)
            else if close >= botV and close <= topV
                if isBull
                    inBullOB    := true
                    bullOBSlRef := botV
                else
                    inBearOB    := true
                    bearOBSlRef := topV
        else
            isCurrBull = isBull != isBreaker
            if isCurrBull
                if close < botV
                    box.delete(array.get(obBoxRefs, i))
                    array.remove(obBoxRefs, i)
                    array.remove(obTop,     i)
                    array.remove(obBot,     i)
                    array.remove(obBar,     i)
                    array.remove(obBull,    i)
                    array.remove(obBreaker, i)
                else if close >= botV and close <= topV
                    inBullOB    := true
                    bullOBSlRef := botV
            else
                if close > topV
                    box.delete(array.get(obBoxRefs, i))
                    array.remove(obBoxRefs, i)
                    array.remove(obTop,     i)
                    array.remove(obBot,     i)
                    array.remove(obBar,     i)
                    array.remove(obBull,    i)
                    array.remove(obBreaker, i)
                else if close >= botV and close <= topV
                    inBearOB    := true
                    bearOBSlRef := topV

if showOBs
    sz = array.size(obBoxRefs)
    if sz > 0
        for i = 0 to sz - 1
            b = array.get(obBoxRefs, i)
            if not na(b)
                box.set_right(b, bar_index)

// ── Fair Value Gaps ───────────────────────────────────────────────────────────
var float[] fvgTop     = array.new_float(0)
var float[] fvgBot     = array.new_float(0)
var int[]   fvgBar     = array.new_int(0)
var bool[]  fvgBull    = array.new_bool(0)
var box[]   fvgBoxRefs = array.new<box>(0)

minSz = fvgMinPct / 100.0
maxSz = fvgMaxPct / 100.0
bullGap    = low    - high[2]
bearGap    = low[2] - high
newBullFvg = low > high[2] and (bullGap / close) >= minSz and (bullGap / close) <= maxSz
newBearFvg = high < low[2] and (bearGap / close) >= minSz and (bearGap / close) <= maxSz

if newBullFvg
    array.unshift(fvgTop,  low)
    array.unshift(fvgBot,  high[2])
    array.unshift(fvgBar,  bar_index[1])
    array.unshift(fvgBull, true)
    newFvgBox = showFvgs ? box.new(
         bar_index[1], low, bar_index, high[2],
         bgcolor=colFvgBull, border_color=colFvgBrdBull,
         border_width=1) : na
    array.unshift(fvgBoxRefs, newFvgBox)
    if array.size(fvgTop) > maxFvgs
        array.pop(fvgTop)
        array.pop(fvgBot)
        array.pop(fvgBar)
        array.pop(fvgBull)
        box.delete(array.pop(fvgBoxRefs))

if newBearFvg
    array.unshift(fvgTop,  low[2])
    array.unshift(fvgBot,  high)
    array.unshift(fvgBar,  bar_index[1])
    array.unshift(fvgBull, false)
    newFvgBox = showFvgs ? box.new(
         bar_index[1], low[2], bar_index, high,
         bgcolor=colFvgBear, border_color=colFvgBrdBear,
         border_width=1) : na
    array.unshift(fvgBoxRefs, newFvgBox)
    if array.size(fvgTop) > maxFvgs
        array.pop(fvgTop)
        array.pop(fvgBot)
        array.pop(fvgBar)
        array.pop(fvgBull)
        box.delete(array.pop(fvgBoxRefs))

// ── FVG Zone Check + Mitigation ───────────────────────────────────────────────
inBullFvg    = false
inBearFvg    = false
bullFvgSlRef = float(na)
bearFvgSlRef = float(na)

nFvg = array.size(fvgTop)
if nFvg > 0
    for i = nFvg - 1 to 0
        topV   = array.get(fvgTop,  i)
        botV   = array.get(fvgBot,  i)
        isBull = array.get(fvgBull, i)
        if isBull and close < botV
            box.delete(array.get(fvgBoxRefs, i))
            array.remove(fvgBoxRefs, i)
            array.remove(fvgTop,  i)
            array.remove(fvgBot,  i)
            array.remove(fvgBar,  i)
            array.remove(fvgBull, i)
        else if not isBull and close > topV
            box.delete(array.get(fvgBoxRefs, i))
            array.remove(fvgBoxRefs, i)
            array.remove(fvgTop,  i)
            array.remove(fvgBot,  i)
            array.remove(fvgBar,  i)
            array.remove(fvgBull, i)
        else
            b = array.get(fvgBoxRefs, i)
            if close >= botV and close <= topV
                if isBull
                    inBullFvg    := true
                    bullFvgSlRef := botV
                    if showFvgs and not na(b)
                        box.set_bgcolor(b, colFvgBullMit)
                else
                    inBearFvg    := true
                    bearFvgSlRef := topV
                    if showFvgs and not na(b)
                        box.set_bgcolor(b, colFvgBearMit)
            else if showFvgs and not na(b)
                box.set_bgcolor(b, isBull ? colFvgBull : colFvgBear)

if showFvgs
    sz = array.size(fvgBoxRefs)
    if sz > 0
        for i = 0 to sz - 1
            b = array.get(fvgBoxRefs, i)
            if not na(b)
                box.set_right(b, bar_index)

// ── Signal Conditions ────────────────────────────────────────────────────────
inBullZone  = inBullFvg or inBullOB
inBearZone  = inBearFvg or inBearOB
rawBullSl   = not na(bullOBSlRef) ? bullOBSlRef : bullFvgSlRef
rawBearSl   = not na(bearOBSlRef) ? bearOBSlRef : bearFvgSlRef
bullSlFinal = rawBullSl - atrVal * atrMult
bearSlFinal = rawBearSl + atrVal * atrMult

bodySize  = math.abs(close - open)
upperWick = high - math.max(close, open)
lowerWick = math.min(close, open) - low

hammer     = close > open and bodySize > 0 and
             lowerWick >= bodySize * wickMult and upperWick <= bodySize * 0.6
bullEngulf = close > open and open[1] > close[1] and close > open[1] and open < close[1]
shootStar  = close < open and bodySize > 0 and
             upperWick >= bodySize * wickMult and lowerWick <= bodySize * 0.6
bearEngulf = close < open and close[1] > open[1] and close < open[1] and open > close[1]

candleLongOk  = useCandle ? (hammer or bullEngulf) : true
candleShortOk = useCandle ? (shootStar or bearEngulf) : true

structLongOk  = scalp ? true : mktStr == "bullish"
structShortOk = scalp ? true : mktStr == "bearish"

buyRaw  = inBullZone and htfBull and structLongOk and
          vwapLongOk  and rsiLongOk  and sessionOk and volOk and
          liqLongOk   and candleLongOk  and newsOk
sellRaw = inBearZone and htfBear and structShortOk and
          vwapShortOk and rsiShortOk and sessionOk and volOk and
          liqShortOk  and candleShortOk and newsOk

var int lastSig = -999
validBuy  = buyRaw  and (bar_index - lastSig > cooldown)
validSell = sellRaw and (bar_index - lastSig > cooldown)
if (validBuy or validSell) and barstate.isconfirmed
    lastSig := bar_index

// ── TP1 Tracking ─────────────────────────────────────────────────────────────

var bool  tp1Long   = false
var bool  tp1Short  = false
var bool  tp1Fired  = false
var int   tp1SetBar = -999

    tp1Short  := false
    tp1Fired  := false
    tp1SetBar := bar_index

if validSell and not na(bearSlFinal)

    tp1Long   := false
    tp1Short  := true
    tp1Fired  := false
    tp1SetBar := bar_index

if not na(tp1Track) and not tp1Fired and bar_index > tp1SetBar
    if tp1Long and high >= tp1Track
        tp1Hit   := true
        tp1Fired := true
        tp1Long  := false
    if tp1Short and low <= tp1Track
        tp1Hit   := true
        tp1Fired := true
        tp1Short := false

    slShort  := true
    slFired  := false
    slSetBar := bar_index

        slFired  := true
        slLong   := false
    if slShort and close > slTrack
        slBreach := true
        slFired  := true
        slShort  := false
