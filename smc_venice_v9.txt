var float  swingHigh    = na
var float  swingLow     = na
var int    swingHighBar = na
var int    swingLowBar  = na
var string mktStr       = "neutral"

if not na(pivHigh)
    swingHigh    := pivHigh
    swingHighBar := bar_index[structLen]
if not na(pivLow)
    swingLow    := pivLow
    swingLowBar := bar_index[structLen]

bosBull   = not na(swingHigh) and close > swingHigh and close[1] <= swingHigh and mktStr == "bullish"
chochBull = not na(swingHigh) and close > swingHigh and close[1] <= swingHigh and (mktStr == "bearish" or mktStr == "neutral")
bosBear   = not na(swingLow)  and close < swingLow  and close[1] >= swingLow  and mktStr == "bearish"
chochBear = not na(swingLow)  and close < swingLow  and close[1] >= swingLow  and (mktStr == "bullish" or mktStr == "neutral")

snapSwingHigh    = swingHigh
snapSwingHighBar = swingHighBar
snapSwingLow     = swingLow
snapSwingLowBar  = swingLowBar

// ── Order Blocks ─────────────────────────────────────────────────────────────
var float[] obTop     = array.new_float(0)
var float[] obBot     = array.new_float(0)
var int[]   obBar     = array.new_int(0)
var bool[]  obBull    = array.new_bool(0)
var box[]   obBoxRefs = array.new<box>(0)

obScanLen = 30

if chochBull or bosBull
    // Clear invalidated bear OBs on bullish structure break
    sz = array.size(obTop)
    if sz > 0
        for i = sz - 1 to 0
            if not array.get(obBull, i)
                box.delete(array.get(obBoxRefs, i))
                array.remove(obBoxRefs, i)
                array.remove(obTop,  i)
                array.remove(obBot,  i)
                array.remove(obBar,  i)
                array.remove(obBull, i)
    mktStr    := "bullish"
    swingHigh := na
    for k = 1 to obScanLen
        if close[k] < open[k]
            obH = obUseWicks ? high[k]  : math.max(open[k], close[k])
            obL = obUseWicks ? low[k]   : math.min(open[k], close[k])
            array.unshift(obTop,  obH)
            array.unshift(obBot,  obL)
            array.unshift(obBar,  bar_index - k)
            array.unshift(obBull, true)
            newBox = showOBs ? box.new(
                 bar_index - k, obH, bar_index, obL,
                 bgcolor=colOBBull, border_color=colOBBrdBull,
                 border_width=2, border_style=line.style_solid,
                 text="OB ▲", text_size=size.tiny,
                 text_color=color.new(colBull, 10),
                 text_halign=text.align_right, text_valign=text.align_top) : na
            array.unshift(obBoxRefs, newBox)
            if array.size(obTop) > maxOBs
                array.pop(obTop)
                array.pop(obBot)
                array.pop(obBar)
                array.pop(obBull)
                box.delete(array.pop(obBoxRefs))
            break

if chochBear or bosBear
    // Clear invalidated bull OBs on bearish structure break
    sz = array.size(obTop)
    if sz > 0
        for i = sz - 1 to 0
            if array.get(obBull, i)
                box.delete(array.get(obBoxRefs, i))
                array.remove(obBoxRefs, i)
                array.remove(obTop,  i)
                array.remove(obBot,  i)
                array.remove(obBar,  i)
                array.remove(obBull, i)
    mktStr   := "bearish"
    swingLow := na
    for k = 1 to obScanLen
        if close[k] > open[k]
            obH = obUseWicks ? high[k]  : math.max(open[k], close[k])
            obL = obUseWicks ? low[k]   : math.min(open[k], close[k])
            array.unshift(obTop,  obH)
            array.unshift(obBot,  obL)
            array.unshift(obBar,  bar_index - k)
            array.unshift(obBull, false)
            newBox = showOBs ? box.new(
                 bar_index - k, obH, bar_index, obL,
                 bgcolor=colOBBear, border_color=colOBBrdBear,
                 border_width=2, border_style=line.style_solid,
                 text="OB ▼", text_size=size.tiny,
                 text_color=color.new(colBear, 10),
                 text_halign=text.align_right, text_valign=text.align_bottom) : na
            array.unshift(obBoxRefs, newBox)
            if array.size(obTop) > maxOBs
                array.pop(obTop)
                array.pop(obBot)
                array.pop(obBar)
                array.pop(obBull)
                box.delete(array.pop(obBoxRefs))
            break

            label.delete(array.shift(structLbls))
    if bosBear or chochBear
        lCol   = bosBear ? color.new(color.gray, 40) : color.new(colChochBear, 20)

// ── Volume Profile (VRVP) ────────────────────────────────────────────────────
var box[] vrvpBoxes = array.new<box>(0)

if showVrvp and barstate.islast
    _sz = array.size(vrvpBoxes)
    if _sz > 0
        for i = 0 to _sz - 1
            box.delete(array.get(vrvpBoxes, i))
    array.clear(vrvpBoxes)

    actualBars = math.min(vrvpBars, bar_index)
    loPrice = low
    hiPrice = high
    for k = 1 to actualBars - 1
        loPrice := math.min(loPrice, low[k])
        hiPrice := math.max(hiPrice, high[k])

    if hiPrice > loPrice
        binSize = (hiPrice - loPrice) / float(vrvpBins)
        binVols = array.new_float(vrvpBins, 0.0)

        for k = 0 to actualBars - 1
            midPx  = (high[k] + low[k]) * 0.5
            binIdx = int(math.floor((midPx - loPrice) / binSize))
            binIdx := math.max(0, math.min(vrvpBins - 1, binIdx))
            array.set(binVols, binIdx, array.get(binVols, binIdx) + volume[k])

        maxVol = array.max(binVols)
        pocIdx = 0
        for b = 0 to vrvpBins - 1
            if array.get(binVols, b) >= array.get(binVols, pocIdx)
                pocIdx := b

        xRight = bar_index + 2
        for b = 0 to vrvpBins - 1
            volV   = array.get(binVols, b)
            volPct = maxVol > 0 ? volV / maxVol : 0.0
            barW   = math.max(1, int(math.round(float(vrvpWidth) * volPct)))
            binLo  = loPrice + float(b) * binSize
            binHi  = binLo + binSize
            isPOC  = b == pocIdx
            bColor = isPOC         ? color.new(#ffd700, 25) :
                     volPct > 0.7  ? color.new(#00c076, 45) :
                     volPct > 0.4  ? color.new(#00c076, 65) :
                                     color.new(#00c076, 82)
            array.push(vrvpBoxes, box.new(xRight - barW, binHi, xRight, binLo,
                 bgcolor=bColor, border_color=color.new(color.gray, 88), border_width=0))
